### **PHASE 3: Strict Arrow Key Loop Math**
**Objective:** Completely rewrite the arrow key slice navigation to strictly respect loop boundaries and eliminate floating-point skipping.

1.  **`App.tsx` (Arrow Key Refactor):**
    * Inside `handleKeyDown`, completely replace the `if (['ArrowUp', ...].includes(e.key))` block with this precision logic:
        ```typescript
        if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.key)) {
          e.preventDefault();
          if (!audioBuffer) return;

          const lBound = loopRange[0];
          const rBound = loopRange[1];
          const activeHits = renderableHits.filter(h => h.state !== 'muted').sort((a,b) => a.sampleIndex - b.sampleIndex);

          // 1. Build an array of times strictly within the loop bounds
          const sliceBoundaries: { time: number, hitIndex?: number }[] = [{ time: lBound }];
          activeHits.forEach(h => {
            const t = h.sampleIndex / sampleRate;
            if (t > lBound + 0.005 && t < rBound - 0.005) {
              sliceBoundaries.push({ time: t, hitIndex: h.sampleIndex });
            }
          });
          sliceBoundaries.push({ time: rBound });

          // 2. Find the exact slice we are currently in
          let currentSliceIdx = 0;
          for (let i = 0; i < sliceBoundaries.length - 1; i++) {
            // using a 5ms epsilon prevents the playhead jitter from skipping slices
            if (currentTime >= sliceBoundaries[i].time - 0.005 && currentTime < sliceBoundaries[i+1].time - 0.005) {
              currentSliceIdx = i;
              break;
            }
          }

          const currentBoundary = sliceBoundaries[currentSliceIdx];

          if (e.key === 'ArrowUp' && currentBoundary.hitIndex !== undefined) {
            toggleHitState(currentBoundary.hitIndex, 'locked');
            setSelectedHitIndices(prev => prev.filter(idx => idx !== currentBoundary.hitIndex));
          } 
          else if (e.key === 'ArrowDown' && currentBoundary.hitIndex !== undefined) {
            toggleHitState(currentBoundary.hitIndex, 'muted');
            setSelectedHitIndices(prev => prev.filter(idx => idx !== currentBoundary.hitIndex));
          } 
          else if (e.key === 'ArrowLeft') {
            const targetIdx = Math.max(0, currentSliceIdx - 1);
            playSlice(sliceBoundaries[targetIdx].time, sliceBoundaries[targetIdx + 1].time);
          } 
          else if (e.key === 'ArrowRight') {
            const targetIdx = Math.min(sliceBoundaries.length - 2, currentSliceIdx + 1);
            playSlice(sliceBoundaries[targetIdx].time, sliceBoundaries[targetIdx + 1].time);
          }
        }
        ```
